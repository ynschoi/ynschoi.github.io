const KINDS = ['heart', 'thumb', 'muscle'];
const CAP = { sn: 270, cj: 320.4, sq: 588 };   // Moti 게임 최대치 (세계신기록 x1.2)
function getBook() {
  const props = PropertiesService.getScriptProperties();
  let id = props.getProperty('SHEET_ID');
  if (!id) {
    id = SpreadsheetApp.create('Random Walk 반응').getId();
    props.setProperty('SHEET_ID', id);
  }
  return SpreadsheetApp.openById(id);
}
function getSheet() {
  const sh = getBook().getSheets()[0];
  if (sh.getLastRow() === 0) sh.appendRow(['시각', '방문자', '이벤트', '내용', '언어', '기기']);
  return sh;
}
function getBoard() {
  const book = getBook();
  let sh = book.getSheetByName('leaderboard');
  if (!sh) {
    sh = book.insertSheet('leaderboard');
    sh.appendRow(['시각', '방문자', '이름', '스내치', '클린앤저크', '백스쿼트', '합계', '버전']);
  }
  return sh;
}
function cut(x, n) { return String(x || '').slice(0, n); }
function kg(x, cap) { const v = Math.round(Number(x) * 2) / 2; return isFinite(v) && v > 0 ? Math.min(v, cap) : 0; }
function json(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function board() {
  const sh = getBoard();
  const last = sh.getLastRow();
  const rows = last > 1 ? sh.getRange(2, 1, last - 1, 8).getValues() : [];
  return rows.map(function (r) { return { v: r[1], name: r[2], sn: r[3], cj: r[4], sq: r[5], total: r[6], t: r[0] }; })
    .sort(function (a, b) { return b.total - a.total; }).slice(0, 30);
}
function doGet(e) {
  const p = (e && e.parameter) || {};
  if (p.op === 'lb' || p.op === 'board') {
    const lock = LockService.getScriptLock();
    lock.waitLock(10000);
    try {
      if (p.op === 'lb' && p.v) {
        const sh = getBoard();
        const sn = kg(p.sn, CAP.sn), cj = kg(p.cj, CAP.cj), sq = kg(p.sq, CAP.sq);
        const row = [new Date(), cut(p.v, 10), cut(p.n, 12) || 'Player', sn, cj, sq, Math.round((sn + cj + sq) * 2) / 2, cut(p.r, 4)];
        const last = sh.getLastRow();
        const ids = last > 1 ? sh.getRange(2, 2, last - 1, 1).getValues() : [];
        let at = -1;
        for (let i = 0; i < ids.length; i++) if (ids[i][0] === row[1]) { at = i + 2; break; }
        if (at > 0) sh.getRange(at, 1, 1, 8).setValues([row]); else sh.appendRow(row);
      }
      return json({ board: board() });
    } finally { lock.releaseLock(); }
  }
  const sh = getSheet();
  if (p.op === 'add' && KINDS.indexOf(p.k) >= 0) {
    sh.appendRow([new Date(), cut(p.v, 10), 'react', p.k, cut(p.l, 4), cut(p.d, 20)]);
  } else if (p.op === 'log') {
    sh.appendRow([new Date(), cut(p.v, 10), cut(p.ev, 30), cut(p.k, 60), cut(p.l, 4), cut(p.d, 20)]);
  }
  const counts = { heart: 0, thumb: 0, muscle: 0 };
  const last = sh.getLastRow();
  if (last > 0) sh.getRange(1, 1, last, 4).getValues().forEach(function (r) {
    if (r[2] === 'react' && counts.hasOwnProperty(r[3])) counts[r[3]]++;
    else if (counts.hasOwnProperty(r[1])) counts[r[1]]++;
  });
  return json(counts);
}
